; WuchangMinimap - overlay settings.
; Plain `key = value` lines; `;` and `#` start a comment. F5 or the F2 panel's "Reload
; map data" applies changes; no restart. The panel writes your edits back here, values
; only. Key names for the hotkeys are listed in the HOTKEY NAMES block below.
; config_wuchang_minimap_dev.txt, when present, is read after this file and overrides it.

; =================================================================================
; ---- PLAYER SETTINGS ----
; =================================================================================
; 1 = run; 0 = inert (no hook, no scans, no map). The F2 panel can switch it off, never on.
mod_enabled = 1

; Draw the minimap disc.
show_minimap = 1

; Chrome colours: neutral | ink.
theme = neutral

; Marker hue set: default | colorblind. Shapes never change.
palette = default

; auto = scale with screen height; a number pins it (1.0 = 1080p sizing).
ui_scale = auto

; Overlay text height in 1080p pixels, times ui_scale. 8..48.
font_size = 20

; custom | top-left | top-right | bottom-left | bottom-right. `custom` obeys the
; minimap_anchor / minimap_offset_* / compass_anchor keys below.
hud_preset = custom

; ---- The minimap ----
; Side length as a fraction of screen height (0.05 .. 0.9).
minimap_size = 0.24

; World units per minimap pixel; smaller = more zoomed in. 1 uu = 1 cm.
minimap_zoom = 26

; round | square
minimap_shape = round

; top-left | top-right | bottom-left | bottom-right, and the offset from it in pixels.
minimap_anchor = top-left
minimap_offset_x = 24
minimap_offset_y = 24

; 0 = north up (the arrow turns); 1 = the map turns with you.
rotate_with_player = 0

; Minimap opacity, 0.1 .. 1.
opacity = 0.92

; Hide the minimap, the compass and the x-ray while any menu is open.
hide_in_menus = 1

; Also draw the storey below (dimmed) and a hint of the one overhead (dimmer).
show_adjacent_floors = 1

; Half-thickness of "my floor" in uu: a surface within this of your feet is opaque.
floor_z_tolerance = 200

; ---- Markers ----
; Draw markers at all.
markers_enabled = 1

; Categories to draw: comma-separated list of shrine, chest, pickup, boss, elite, enemy,
; npc, note, door, ladder, lift, fog_gate, hidden, other - or `all` / `none`.
markers_categories = shrine,chest,pickup,boss,elite,npc,note,door,ladder,lift,fog_gate,hidden,other

; Found markers: 0 = draw dimmed and hollow, 1 = hide. Shrines always stay on the map.
markers_hide_found = 1

; Glyph radius in minimap pixels.
markers_size = 6.5

; 1 = keep off-map markers on the rim, drawn smaller; 0 = drop them.
markers_clamp_to_edge = 0

; ---- The full map ----
; Zoom the map opens at, in uu per screen pixel; closing it writes back where you left it.
map_zoom = 30

; Marker glyph radius on the full map, in screen pixels.
map_marker_size = 8.0

; Allow the controller to open the map (polls XInput).
map_gamepad = 1

; ---- The x-ray highlight ----
; Draw nearby markers through walls while armed.
highlight_enabled = 1

; toggle = press on, press off; hold = on only while held.
highlight_mode = toggle

; The key that arms the highlight.
highlight_key = TAB

; Pad route: 1 enables the chord. Buttons joined with `+`, from LB, RB, LT, RT, A, B, X,
; Y, BACK, START, LS, RS, UP, DOWN, LEFT, RIGHT; `none` disables it.
highlight_gamepad = 1
highlight_pad_chord = LB+RB

; How far a marker may be from the player, in uu.
highlight_radius = 3000

; Categories to highlight; same names as markers_categories, or `all` / `none`.
highlight_categories = chest,pickup,shrine,boss,npc,note

; Draw the name and distance next to each glyph.
highlight_labels = 1

; 1 = also highlight collected loot, dimmed.
highlight_show_found = 0

; Glyph radius on screen, in pixels.
highlight_size = 7.0

; 1 = colour a pickup by item-quality tier instead of category: 1 Equipment (pink beam),
; 2 Key (gold beam). Tier 0 Common keeps its category colour.
xray_rarity_colors_enabled = 1

; 1 = use the tier colours on the minimap, the full map and the compass pips too.
markers_rarity_tint = 0

; ---- The compass strip ----
; Draw the heading strip.
compass_enabled = 1

; Which edge the strip hangs off: top | bottom.
compass_anchor = top

; Strip width as a fraction of screen width, and the distance from that edge in pixels.
compass_width = 0.340
compass_offset_y = 18

; Degrees the whole strip covers; 360 makes it a full ring.
compass_span_deg = 120

; Strip opacity, 0.1 .. 1.
compass_opacity = 0.90

; Categories that get a bearing pip.
compass_categories = shrine,boss,elite,fog_gate

; Write the distance in metres under each pip.
compass_pip_labels = 1

; ---- Hotkeys ----
; HOTKEY NAMES: F1..F5, F7, F8, A..Z, 0..9, TAB, SPACE, ENTER, BACKSPACE, UP / DOWN /
; LEFT / RIGHT, INSERT / DELETE / HOME / END / PAGEUP / PAGEDOWN, NUM0..NUM9, NUMPLUS /
; NUMMINUS / NUMMUL / NUMDIV / NUMDOT, MOUSE3 / MOUSE4 / MOUSE5, LALT / RALT / ALT /
; LSHIFT / RSHIFT / SHIFT / LCTRL / RCTRL / CTRL, or `none` for unbound. One modifier may
; prefix a key (`ctrl+m`). F6 and F9..F12 are rejected and fall back to the default.
; The F2 panel's Keys tab rebinds every key by pressing the new one.
; Pad buttons that open and close the full map and the settings panel, joined with `+`;
; `none` disables one. The map chord needs map_gamepad = 1; analogue triggers are out.
map_pad_open_chord = BACK+Y
panel_pad_open_chord = BACK+RS

; The settings panel, the full map, recentring the full map, and reloading config and data.
panel_key = F2
map_key = M
map_recenter_key = R
reload_key = F5

; Steps the minimap zoom through minimap_zoom_presets.
zoom_key = N

; Copies the full map to the clipboard as a bitmap while the full map is open.
screenshot_key = C

; Sets a waypoint on the nearest marker you have not found yet. Unbound as shipped - the
; game itself uses G, so pick a free key.
waypoint_nearest_key = none

; =================================================================================
; ---- ADVANCED ----
; =================================================================================
; Correct as shipped; change one to answer a symptom.

; ---- When the overlay is allowed on screen ----
; Show the overlay only while the camera is on the player pawn (keeps it out of menus
; and cutscenes).
require_pawn_view = 1

; Hide the minimap if no position has been published for this many ms.
state_stale_ms = 1000

; How long a validated gameplay pawn must be held before the minimap may appear, in ms.
min_visible_after_state_ok_ms = 600

; Delay before the minimap comes back after a menu closes, in ms. Hiding is immediate.
menu_close_show_delay_ms = 50

; ---- Height slicing ----
; How many times a second the CPU re-cuts the minimap window (2..30).
slice_hz = 12

; Time constant of the smoothing on your feet Z, in ms.
feet_z_smooth_ms = 100

; uu subtracted from the pawn's capsule centre to get feet Z.
player_z_offset = 90

; ---- Marker sweep and tracker tuning ----
; Run the live actor sweep. 0 = static positions only, no collected or opened state.
markers_live = 1

; Draw only the current chapter's static markers; 0 paints other chapters' markers too.
markers_filter_chapter = 1

; Sweep pacing: rounds per second (1..10), object slots per pump (512..131072) - the
; frame-cost dial - and the minimum ms between pumps (1..500).
markers_rounds_per_sec = 1
markers_scan_chunk = 8192
markers_scan_period_ms = 8

; How dim a found marker is, as a multiple of `opacity`.
markers_found_alpha = 0.30

; Hard cap on markers drawn per frame, nearest first.
markers_max_draw = 400

; How long after the last change the found file is written, in ms.
found_save_debounce_ms = 2000

; A loaded-but-unseen chest or pickup counts as collected after this many rounds in a
; row, in these categories. Undo a mark by clicking the marker on the full map.
markers_absence_rounds = 2
markers_absence_categories = chest,pickup

; 1 = count a boss as defeated when its arena's respawn point is unlocked in the save.
; May mark a boss you fought and did not beat; 0 undoes it completely.
boss_defeat_from_save = 1

; ---- Minimap look ----
; The ladder zoom_key steps through, in uu per minimap pixel: 1 to 8 numbers between 2
; and 400, in any order.
minimap_zoom_presets = 13, 26, 52

; The dark disc the map is drawn on: opacity, then "R G B".
minimap_backdrop = 0.86
minimap_backdrop_color = 6 9 13

; The ring around the minimap: "R G B", then opacity.
minimap_frame_color = 168 176 186
minimap_frame_alpha = 0.85

; Floor on the minimap's side length, in pixels.
minimap_min_px = 72

; The player arrow: a fraction of the minimap's side, and a floor in pixels.
minimap_arrow_frac = 0.055
minimap_arrow_min_px = 8

; Waypoint glyph radius, as a multiple of markers_size.
waypoint_size_scale = 1.05

; ---- Full map tuning ----
; Zoom limits, in uu per screen pixel (min = most zoomed in).
map_zoom_min = 4
map_zoom_max = 240

; Zoom multiplier per wheel notch (1.01 .. 2.0).
map_zoom_factor = 1.15

; Keyboard / stick pan speed, in screen pixels per second.
map_pan_speed = 900

; Border around the map as a fraction of screen height, and the opacity behind it.
map_margin = 0.045
map_backdrop = 0.86

; Hard cap on markers drawn per frame on the full map.
map_markers_max_draw = 4000

; How far one press of the floor adjustment moves the height slice, in uu.
map_floor_step = 200

; The map's own slice texture: width in pixels, and the cap on re-cuts per second.
map_slice_px = 768
map_slice_hz = 6

; Stick deadzone, as a fraction of full deflection.
map_gamepad_deadzone = 0.22

; ---- X-ray tuning ----
; Hard cap on glyphs per frame, nearest first.
highlight_max_draw = 60

; Hard cap on labels (name + distance), 0 .. 40. They go to the nearest markers.
highlight_labels_max = 12

; Alpha at the camera and at highlight_radius; the fade between them is linear, and far
; is never above near.
highlight_alpha_near = 1.00
highlight_alpha_far = 0.25

; 1 = anything off screen or behind the camera gets an arrow on the screen edge.
highlight_edge_arrows = 1

; How often the camera is re-read while the highlight is armed, in Hz.
highlight_camera_hz = 60

; One RRGGBB per item-quality tier, in tier order (Common, Equipment, Key). A `#` prefix
; and the 3-digit short form are accepted.
xray_rarity_colors = ADAFDA, DAADC5, DAD6AD

; ---- Compass tuning ----
; Strip height in pixels.
compass_height = 26

; 1 = a filled plate behind the strip; 0 = ticks and letters only, each shadowed.
compass_plate = 1

; How far away a marker may be to get a pip, in uu.
compass_marker_distance = 15000

; The waypoint is never culled: off the strip it clamps to the edge with an arrow.
compass_show_waypoint = 1

; Minor-tick spacing in degrees. A tick at a multiple of 45 gets a label, one at a
; multiple of 90 gets its cardinal letter.
compass_tick_step_deg = 15

; Hard cap on marker pips, nearest first.
compass_max_pips = 32

; A glyph and a pip get an up / down arrow once the marker is this far off your own
; height, in uu. 0 turns the arrows off.
compass_pip_height_uu = 300

; ---- Text ----
; Full path to a .ttf or .otf for the whole overlay. `none`, empty or an unreadable file
; falls back to ImGui's built-in font.
ui_font = C:\Windows\Fonts\segoeui.ttf
