; WuchangMinimap - overlay settings.
;
; Plain `key = value` lines; `;` and `#` start a comment. Written by the F2 settings
; panel ("Save settings") and re-read by F5 or "Reload settings + maps", so the game
; does not have to be restarted for a change here.
;
; HOTKEYS: only F1..F5, F7 and F8 are accepted. Anything else falls back to the
; default, because on this machine
;   F6  toggles RenoDX / DLSS 5 Neural Rendering (it ignores modifiers, and pressing
;       it in-world hung the GPU on 2026-09-02)
;   F9  / F11 are engine screenshot / fullscreen binds
;   F10 opens the game console (UE4SS ConsoleEnablerMod)
;   F12 is the Steam screenshot key

; Master switch. 0 = install no DX12 hooks at all.
enabled = 1

; Draw the minimap. Turn off to keep only the F2 panel.
show_minimap = 1

; Minimap side length as a fraction of the screen height (0.05 .. 0.9).
minimap_size = 0.24

; World units per minimap pixel. Smaller = more zoomed in. 1 uu = 1 cm, so 26 shows
; roughly a 70 m square on a 270 px minimap.
minimap_zoom = 26

; round | square
minimap_shape = round

; top-left | top-right | bottom-left | bottom-right
minimap_anchor = top-left
minimap_offset_x = 24
minimap_offset_y = 24

; 0 = north up (the map never turns, the arrow does); 1 = the map turns with you.
rotate_with_player = 0

; 0.1 .. 1.0
opacity = 0.92

; Hide the minimap while any menu is open (inventory / pause / shrine map).
hide_in_menus = 1

; Hide the minimap unless the camera's view target is the player pawn. This is what
; keeps it off the main menu (the Lobby view target is a CineCameraActor) and out of
; cutscenes.
require_pawn_view = 1

; Hide the minimap if the game-thread reader has not published a position in this
; many milliseconds.
state_stale_ms = 1000

; Grace period: how long the reader must have had a validated gameplay pawn
; (class BP_CombatCharacter_Player*, live in GUObjectArray) before the minimap is
; allowed to appear. Stops it flashing during a level load / right after Continue.
min_visible_after_state_ok_ms = 600

; A menu closing is not a level transition, so showing the minimap again waits only
; this long (hiding is immediate - the game thread re-tests the cached in-viewport
; menu roots on every 10 Hz pump).
menu_close_show_delay_ms = 150

; ---------------------------------------------------------------------------------
; Floor (Z) awareness
; ---------------------------------------------------------------------------------
; The map ships one pre-rendered layer per floor (maps\chapter1\small_f<i>.png) plus
; a per-cell table of walkable surface bands (maps.json "floor_grid"). Every frame the
; overlay looks up the band the player is standing in and draws that band's layer, so
; standing in the dungeon under a temple shows the dungeon, not the roof.

; Also draw the storey below (dimmed, cool) and above (dimmer, warm).
show_adjacent_floors = 1

; Opacity of the storey below, as a multiple of `opacity`. The storey above uses 60 %
; of this. 0 = draw only the floor you are on.
adjacent_floor_opacity = 0.25

; Slack around a band's Z range, in uu (1 uu = 1 cm), before the player counts as
; "between floors". 150 = 1.5 m.
floor_z_tolerance = 150

; How much closer another band has to be before the minimap switches to it. Stops the
; picture flickering between two layers halfway up a staircase.
floor_hysteresis = 100

; The pawn's reported location is its capsule centre, about 90 uu above the navmesh it
; stands on; this is subtracted before the band lookup.
player_z_offset = 90

; If the player is off the floor grid (off-navmesh, an unmapped area), the last floor
; is kept for this long before the fallback takes over.
floor_fallback_hold_ms = 3000

; The fallback: 0 = draw every floor layer at once (free - the layers are already
; resident). 1 = load and draw the Z-shaded composite small.png instead, which costs
; another 82 MB of VRAM on top of the layers' ~104 MB for Chapter 1.
fallback_use_composite = 0

; Show the world position / uv / chapter / floor readout in the F2 panel.
debug_readout = 1

; Open the settings panel as soon as the mod starts. Development aid only - it is how
; the panel is verified without a keypress. Leave at 0 for normal play.
debug_show_panel_on_start = 0

; Settings panel toggle, and config+map reload.
panel_key = F2
reload_key = F5
