; WuchangMinimap - overlay settings.
;
; Plain `key = value` lines; `;` and `#` start a comment. Written by the F2 settings
; panel ("Save settings") and re-read by F5 or "Reload settings + maps", so the game
; does not have to be restarted for a change here.
;
; HOTKEYS: F1..F5, F7, F8, any single letter (A..Z) or digit (0..9), TAB, SPACE, or
; one of LALT / RALT / ALT / LSHIFT / RSHIFT / SHIFT / LCTRL / RCTRL / CTRL.
; F6 and F9..F12 are REJECTED and fall back to the default, because on this machine
;   F6  toggles RenoDX / DLSS 5 Neural Rendering (it ignores modifiers, and pressing
;       it in-world hung the GPU on 2026-09-02)
;   F9  / F11 are engine screenshot / fullscreen binds
;   F10 opens the game console (UE4SS ConsoleEnablerMod)
;   F12 is the Steam screenshot key
; Shipped defaults: F2 settings panel, M full map, R recentre the map, F5 reload,
; and LALT HELD for the x-ray highlight.

; Master switch. 0 = install no DX12 hooks at all.
enabled = 1

; Draw the minimap. Turn off to keep only the F2 panel.
show_minimap = 1

; Minimap side length as a fraction of the screen height (0.05 .. 0.9).
minimap_size = 0.24

; World units per minimap pixel. Smaller = more zoomed in. 1 uu = 1 cm, so 26 shows
; roughly a 70 m square on a 270 px minimap.
minimap_zoom = 26

; round | square
minimap_shape = round

; top-left | top-right | bottom-left | bottom-right
minimap_anchor = top-left
minimap_offset_x = 24
minimap_offset_y = 24

; 0 = north up (the map never turns, the arrow does); 1 = the map turns with you.
rotate_with_player = 0

; 0.1 .. 1.0
opacity = 0.92

; Hide the minimap while any menu is open (inventory / pause / shrine map).
hide_in_menus = 1

; Hide the minimap unless the camera's view target is the player pawn. This is what
; keeps it off the main menu (the Lobby view target is a CineCameraActor) and out of
; cutscenes.
require_pawn_view = 1

; Hide the minimap if the game-thread reader has not published a position in this
; many milliseconds.
state_stale_ms = 1000

; Grace period: how long the reader must have had a validated gameplay pawn
; (class BP_CombatCharacter_Player*, live in GUObjectArray) before the minimap is
; allowed to appear. Stops it flashing during a level load / right after Continue.
min_visible_after_state_ok_ms = 600

; A menu closing is not a level transition, so showing the minimap again waits only
; this long (hiding is immediate - the game thread re-tests the cached in-viewport
; menu roots on every 10 Hz pump).
menu_close_show_delay_ms = 150

; ---------------------------------------------------------------------------------
; Height slicing - what the minimap actually draws
; ---------------------------------------------------------------------------------
; The map ships a MULTI-SURFACE HEIGHT MAP: four 16-bit PNGs
; (maps\chapter1\small_z0..z3.png) holding the Z of up to four stacked walkable
; surfaces per pixel, lowest first. Roughly 12 times a second the overlay cuts the
; window around the player out of them on the CPU and colours each pixel by how far
; that surface is from the player's feet:
;
;   |surfaceZ - feetZ| <= floor_z_tolerance   ->  opaque  (the floor you are on)
;   the nearest surface below, within fade    ->  adjacent_floor_opacity
;   the nearest surface above, within fade    ->  adjacent_floor_opacity x 0.6
;   nothing                                   ->  transparent
;
; So standing in the dungeon under a temple shows the dungeon, and the temple interior
; is one coherent floor plan instead of several stacked storeys blended together.
; (This replaced the old per-pixel surface-ORDINAL layers, which were exactly
; separable but unreadable: the runtime could only guess which ordinal your storey
; was. `floor_hysteresis` and `floor_fallback_hold_ms` no longer exist.)

; Also draw the surfaces below (dimmed) and above (dimmer).
show_adjacent_floors = 1

; Opacity of the surface below, as a multiple of `opacity`. The one above uses 60 % of
; this. 0 = draw only the floor you are on.
adjacent_floor_opacity = 0.25

; Half-thickness of "my floor", in uu (1 uu = 1 cm). Any surface whose Z is within this
; of your feet is drawn opaque, so a sloped courtyard or a flight of stairs stays one
; floor. 200 = 2 m.
floor_z_tolerance = 200

; How far below / above your feet a surface is still drawn (dimmed) before it is
; dropped entirely. 800 = 8 m, i.e. roughly one storey.
floor_fade_uu = 800

; Height gradient. The fill is shaded by
;     lum = 1 + floor_gradient_strength * clamp((surfaceZ - feetZ) / span, -1, +1)
; with span = floor_z_tolerance for your own floor and floor_fade_uu for the dimmed
; ones, so ramps and staircases read as a gentle gradient instead of a flat
; silhouette. 0 = a flat fill.
floor_gradient_strength = 0.18

; The walkable fill colour, "R G B" (0..255 each). A light warm grey that reads on this
; game's dark scenes.
floor_base_color = 214 208 196

; How many times a second the CPU re-cuts the window (2..30). 12 is invisible in motion
; and costs ~1-3 ms per update on one core; the window is cut with margin, so a skipped
; update never shows an edge.
slice_hz = 12

; Time constant of the EMA on your feet Z, in ms. Stops a jump or a single step from
; snapping the whole picture to another floor.
feet_z_smooth_ms = 100

; The pawn's reported location is its capsule centre, about 90 uu above the navmesh it
; stands on; this is subtracted to get feet Z.
player_z_offset = 90

; Also load the Z-shaded composite small.png (+82 MB of VRAM for Chapter 1). It merges
; every storey into one picture, so it is only useful as the fallback for a chapter
; whose height maps are missing - which the runtime loads on its own anyway.
fallback_use_composite = 0

; Show the world position / uv / chapter / slice readout AND the
; "hidden because: <reason>" line in the F2 panel.
debug_readout = 1

; Open the settings panel as soon as the mod starts. Development aid only - it is how
; the panel is verified without a keypress. Leave at 0 for normal play.
debug_show_panel_on_start = 0

; ---------------------------------------------------------------------------------
; Markers
; ---------------------------------------------------------------------------------
; Markers come from two places and are merged by a stable id:
;
;   the STATIC database  markers\<chapter>.json, produced offline from the cooked
;                        levels, so markers show up in areas you have never visited;
;   the LIVE sweep       one FindAllOf per game-thread tick, cycling through the
;                        marker classes, which supplies the actual state of every
;                        actor that is currently streamed in - a chest's `Used`, a
;                        door's `DoorOpen`, a pickup's `dying` / parked-at-origin,
;                        and the shrine's own game id.
;
; Enemies are live only and are never written to the collection tracker.

; Draw markers at all.
markers_enabled = 1

; Run the live actor sweep on the game thread. Turning this off leaves only the
; static database (positions, but no collected / opened state).
markers_live = 1

; Draw only the markers of the chapter you are actually in.
;
; The static database is one flat set of all six chapters' 3601 markers, and the
; chapters' world extents overlap badly - chapter 4's bounds cover nearly all of
; chapter 1's - so with this off you get other chapters' shrines and chests painted
; on top of the current map. The chapter is named by the streamed level packages, not
; by your position (a position cannot tell chapter 1 from chapter 4). Until it has
; been recognised, nothing is filtered. Live actors are in the current world by
; definition and are never filtered. Set to 0 only to debug the marker database.
markers_filter_chapter = 1

; How the live sweep works, and its three dials.
;
; The sweep walks the engine's object array ONCE per full pass, a slice at a time, and
; classifies each object by its blueprint class. A "pump" is one slice; a "round" is a
; complete pass, after which the marker set is republished.
;
;   markers_rounds_per_sec  (1..10)      how often a full pass may START. A pass that
;                                        finishes early idles until its next turn, so
;                                        this is a ceiling, not a target.
;   markers_scan_chunk      (512..131072) object slots visited per pump. This is the
;                                        frame-cost dial: more slots = a fresher marker
;                                        set and a costlier pump.
;   markers_scan_period_ms  (1..500)     minimum milliseconds between pumps.
;
; The F2 panel prints both numbers you need to tune these: `scan pump` (what one pump
; costs the game thread - keep it under ~1 ms) and `round` (how long a full pass took
; and how many object slots it covered). Halve the chunk if pumps are expensive; the
; round simply takes proportionally longer.
markers_rounds_per_sec = 1
markers_scan_chunk = 8192
markers_scan_period_ms = 8

; Which categories to draw. A comma-separated list of
;   shrine, chest, pickup, boss, elite, enemy, npc, merchant, door, ladder, lift,
;   fog_gate, hidden, other
; or `all` / `none`. The F2 panel edits the same setting. Enemies are off by default
; because the sweep only refreshes them once a second, so they lag while they move.
markers_categories = shrine,chest,pickup,boss,elite,npc,merchant,door,ladder,lift,fog_gate,hidden,other

; Found (collected / opened) markers: 0 = draw them dimmed, 1 = hide them entirely.
markers_hide_found = 0

; How dim a found marker is, as a multiple of `opacity`.
markers_found_alpha = 0.30

; Glyph radius in minimap pixels.
markers_size = 6.5

; Keep markers that are outside the minimap on its rim, drawn smaller, instead of
; dropping them.
markers_clamp_to_edge = 0

; Hard cap on markers drawn per frame, nearest first. Safety valve only.
markers_max_draw = 400

; ---------------------------------------------------------------------------------
; Collection tracker
; ---------------------------------------------------------------------------------
; wuchang_minimap_found.txt (next to this file) holds one stable id per line and is
; rewritten whenever the live sweep auto-marks something new. It is plain text and
; hand-editable; edits are picked up on the next reload.

found_tracker = 1

; How long after the last change the file is written, in ms. Debounced so a room
; full of pickups is one write, not thirty.
found_save_debounce_ms = 2000

; ---------------------------------------------------------------------------------
; The full map (press map_key - shipped default M)
; ---------------------------------------------------------------------------------
; The same height-sliced asset the minimap draws, at map scale: a north-up window over
; the chapter, pannable and zoomable, with every marker on it. While it is open the
; minimap is hidden and the mod takes the mouse and the keyboard; closing it gives both
; back on the same frame.
;
;   drag, WASD or the arrows        pan
;   mouse wheel, + / -              zoom
;   ctrl+wheel, Q / E, PageUp/Down  move the height slice up / down a floor
;   map_recenter_key (R)            back to the player, floor offset cleared
;   left-click a marker, or F       toggle "found" by hand (writes the tracker)
;   right-click, or Space           set the waypoint (also shown on the minimap)
;   map_key or Esc                  close
;   gamepad                         left stick pan, triggers / right stick zoom,
;                                   LB/RB floor, A waypoint, X found, Y recentre,
;                                   B close
;
; Zoom is world units per SCREEN pixel - the same unit as minimap_zoom above, so the
; numbers are directly comparable (26 is the minimap's default). Chapter 1 is about
; 450 m across, so ~55 fits the whole chapter on a 900 px canvas.

; Zoom the map opens at, and the limits it can be taken to.
map_zoom = 30
map_zoom_min = 4
map_zoom_max = 240

; Zoom multiplier per wheel notch (1.01 .. 2.0).
map_zoom_factor = 1.15

; Keyboard / stick pan speed, in SCREEN pixels per second (so it feels the same at
; every zoom).
map_pan_speed = 900

; Border left around the map, as a fraction of the screen height, and how dark the
; backdrop behind it is.
map_margin = 0.045
map_backdrop = 0.86

; Marker glyph radius on the full map, in screen pixels, and the hard cap on how many
; are drawn per frame.
map_marker_size = 8.0
map_markers_max_draw = 4000

; How far one press of the floor adjustment moves the height slice, in uu. The slice
; itself uses floor_z_tolerance / floor_fade_uu, exactly like the minimap.
map_floor_step = 200

; 1 = ignore the height slice and draw every walkable surface (the current storey stays
; opaque, everything else is dimmed). Useful for route planning, noisy in a temple.
map_show_all_floors = 0

; The dynamic texture the map is cut into. The cut costs roughly
; (width x height x surfaces) reads, so 768 lands around 5-10 ms - which is why it only
; runs when the view actually changed (pan out of the cut region, zoom, floor, or a big
; player move), never more than map_slice_hz times a second.
map_slice_px = 768
map_slice_hz = 6

; Gamepad (XInput). Only polled while the map is open, so a controller costs nothing
; during play. The deadzone is a fraction of full stick deflection.
map_gamepad = 1
map_gamepad_deadzone = 0.22

; Remember the waypoint across sessions in wuchang_minimap_waypoint.txt (next to this
; file). It is three plain `key = value` lines and is hand-editable.
map_waypoint_persist = 1

; ---------------------------------------------------------------------------------
; Hold-key x-ray highlight
; ---------------------------------------------------------------------------------
; HOLD highlight_key (or the gamepad chord) in-world: every marker of the categories
; below that is still uncollected and within highlight_radius is drawn at its position
; ON SCREEN - category glyph, name and distance in metres - through walls, fading with
; distance. It is a HOLD, not a toggle, so there is no state that can get stuck, and it
; hides with the minimap in menus and cutscenes.
;
; It needs the camera, which is read on the game thread from the PlayerCameraManager's
; camera cache. The F2 panel's "X-ray highlight" block prints which route is in use and
; where the POV block was found - screenshot that block if the labels are misplaced.

highlight_enabled = 1

; Any hotkey name (see the list at the top of this file). LALT is the shipped default:
; it is a hold, so it wants a key the hand is already resting on.
highlight_key = LALT

; The second route, for a controller. Join buttons with `+`; known names are
; LB, RB, LT, RT, A, B, X, Y, BACK, START, LS, RS, UP, DOWN, LEFT, RIGHT.
; `none` disables the chord. XInput is only polled while a chord is configured.
highlight_gamepad = 1
highlight_pad_chord = LB+RB

; How far a marker may be from the PLAYER (not the camera) to be highlighted, in uu.
; 1 uu = 1 cm, so 3000 = 30 m.
highlight_radius = 3000

; Which categories. Same names as markers_categories: shrine, chest, pickup, boss,
; elite, enemy, npc, merchant, door, ladder, lift, fog_gate, hidden, other (or
; `all` / `none`). Chests and pickups is the "loot I have not taken yet" set.
highlight_categories = chest,pickup

; 1 = also highlight things already collected / opened (dimmed).
highlight_show_found = 0

; Hard cap per frame, nearest first.
highlight_max_draw = 60

; Alpha at the camera and at the radius; the fade between them is linear.
highlight_alpha_near = 1.00
highlight_alpha_far = 0.25

; Glyph radius on screen, in pixels, and whether the name + distance is drawn.
highlight_size = 7.0
highlight_labels = 1

; 1 = anything off screen or behind the camera gets an arrow on the screen edge
; pointing the way to turn.
highlight_edge_arrows = 1

; How often the game thread re-reads the camera WHILE THE KEY IS HELD. The read is a
; few dozen bytes at a cached offset, so this is cheap; it only has to keep up with how
; fast the camera can swing. Nothing is read at all when the key is up and the compass
; is off.
highlight_camera_hz = 60

; ---------------------------------------------------------------------------------
; The compass strip
; ---------------------------------------------------------------------------------
; A heading strip across the top of the screen: N / NE / E ... plus 15-degree ticks,
; with bearing pips for the waypoint and for nearby markers of the categories below.
; The heading is the camera's yaw (the player's yaw while the camera reader is warming
; up). It is hidden by exactly the same rules as the minimap and disappears while the
; full map is open.

compass_enabled = 1

; Strip width as a fraction of the screen width, distance from the top of the screen
; in pixels, and the strip's height in pixels.
compass_width = 0.420
compass_offset_y = 18
compass_height = 26

; How many degrees the whole strip covers. 120 is a comfortable "what is in front of
; me"; 360 turns it into a full ring.
compass_span_deg = 120

compass_opacity = 0.90

; Which markers get a bearing pip, and how far away they may be (uu).
compass_categories = shrine,boss,elite,fog_gate
compass_marker_distance = 15000

; The waypoint is never culled: off the strip it clamps to the edge with an arrow.
compass_show_waypoint = 1

; Settings panel, full map, recentre the full map, and config + map + marker reload.
panel_key = F2
map_key = M
map_recenter_key = R
reload_key = F5
