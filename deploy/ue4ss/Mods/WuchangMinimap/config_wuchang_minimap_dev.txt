; WuchangMinimap - DEVELOPER settings.
;
; Not part of the release. Read only when it exists, after
; config_wuchang_minimap.txt, so anything here overrides that file. Delete it to go
; back to the built-in defaults. The F2 panel's "Debug" tab (visible while
; debug_readout = 1) edits these keys and saves them back here.
;
; Only `srv_heap_size` and `navmesh_dump` need a restart. Everything else is picked up by
; F5 or by the 1 Hz watch on both files' timestamps.

; ---------------------------------------------------------------------------------

; ---------------------------------------------------------------------------------

; "hidden because: <reason>" line in the F2 panel, and the Debug tab itself.
debug_readout = 1

; Open the settings panel as soon as the mod starts.
debug_show_panel_on_start = 0

; Arm the runtime dtNavMesh dumper, which scans engine memory and writes tile JSON into

navmesh_dump = 0

; How much goes into wuchang_minimap.log and UE4SS.log:
;   normal   startup, what loaded, chapter and save-slot changes, warnings, errors,

;   trace    adds everything else, including a marker census every two seconds
log_level = normal

; Rewrite wuchang_minimap_last_stage.txt at every overlay stage (hooks installed,

crash_breadcrumb = 1

; Show a 10-second toast naming the panel, map and highlight keys on the first launch
; after installing. Delete wuchang_minimap_firstrun.txt to see it again.
first_run_toast = 1

; ---------------------------------------------------------------------------------
; MAP BACKGROUND AND DPI
; ---------------------------------------------------------------------------------
; The map assets mark every surface the game's own navigation data says a player can
; walk to. What to do with the rest - wall tops, roof ridges, ledges outside the arena:
;   hide  the map draws only ground you can reach
;   dim   it is drawn one step fainter than ground you can reach
;   show  it is drawn like any other ground
; Map assets built before this existed have no such marks, and everything in them is
; treated as reachable whatever this says.
map_unreachable = hide

; ---------------------------------------------------------------------------------
; HEIGHT SHADING
; ---------------------------------------------------------------------------------
; A pixel's COLOUR is its surface's absolute world Z on a two-colour ramp - one ramp for
; every storey, so a height is the same colour wherever it is and the cut reads as a
; floor plan. The marker at the centre says which storey is yours. The F2 panel's Debug
; tab edits all of it live, under "Developer settings > Height shading".

; The ramp's two ends, "R G B" (0..255 each): low ground and high ground.
shade_lo_color = 88 84 78
shade_hi_color = 244 240 232

; The curve along the ramp: t -> pow(t, gamma). Below 1 lifts the low ground, above 1
; holds the contrast for the peaks. 1 = linear.
shade_gamma = 0.80

; Opacity of the storey below you, and of a ledge or upper terrain overhead. Your own
; storey is always fully opaque; at 1.00 so are these, and height alone separates them.
; show_adjacent_floors = 0 in the player config forces both to 0.
shade_below_alpha = 1.00
shade_above_alpha = 1.00

; How far ABOVE YOUR FEET a surface may sit and still be drawn, in uu - a ledge, or the
; upper terrain a ramp leads to. Anything higher is dropped, so a ceiling never covers
; you. One storey, because a deck over solid floor of yours is hidden by that floor
; anyway and only its edges would show. 0 = never draw anything above you. THE MINIMAP
; ONLY: the full map draws the nearest storey at or above your feet, however high.
shade_above_band_uu = 600

; The ramp's ends are percentiles of the Z of the pixels a cut actually draws, not their
; min and max: one deep pit or one high gallery must not push every playable storey into
; two colour levels. This is the LOW percentile; the high one is 100 minus it. 0 = the
; drawn min..max.
shade_range_pct_lo = 3

; The narrowest span the minimap will stretch the ramp over, in uu, so flat ground does
; not explode to full contrast...
shade_min_range_uu = 400

; ...and the time constant of the easing on that span, in ms, so walking up a stair does
; not make the picture breathe. A teleport resets it. The full map ignores both: it
; equalises its own cut instead (shade_map_equalize).
shade_range_smooth_ms = 400

; 1 = the FULL MAP spends its ramp on area instead of on height: a pixel's t is the CDF
; of the Z its cut drew, so every tenth of the ramp holds a tenth of the drawn pixels and
; a stretch of Z nothing sits at costs no contrast. 0 = the linear percentile ramp, to
; compare the two. The minimap is always linear - its window already tracks one storey.
shade_map_equalize = 1

; 1 = scale minimap_zoom and map_zoom with ui_scale, so one config shows the same area
; of the world at every resolution. 0 = uu per pixel is literal.
zoom_dpi_scaled = 1

; Which collection file: auto | shared | <name>. auto = one file per save slot,
; wuchang_minimap_found_<slot>.txt, seeded once from the shared file. shared = the
; single wuchang_minimap_found.txt. <name> = wuchang_minimap_found_<name>.txt.
found_profile = auto

; Also load the Z-shaded composite small.png (+82 MB of VRAM for Chapter 1). It is the
; fallback for a chapter with no height planes, which the runtime loads on its own.
fallback_use_composite = 0

; Opacity of that composite, as a multiple of `opacity`.
minimap_composite_alpha = 0.85

; ---------------------------------------------------------------------------------
; THE GAME-STATE READER (game thread)
; ---------------------------------------------------------------------------------
; How often the pawn's location and yaw are read, in ms.
reader_position_period_ms = 100

; How often a missing pawn / controller is re-found, in ms.
reader_resolve_period_ms = 500

; The full menu-widget sweep, in ms. Widgets already known are re-tested on every
; pump, which is what makes the minimap hide the instant a menu opens.
reader_widget_sweep_period_ms = 250

; The sweep is a whole-object-array walk, and it only has to DISCOVER a menu root that
; has never been seen this session. Its period doubles up to this maximum while
; nothing new turns up, and drops back to the fast cadence for _warm_ms after any
; menu-state flip, teleport or world change.
reader_widget_sweep_max_period_ms = 2000
reader_widget_sweep_warm_ms = 2000

; After a pawn or world change, no blueprint getter is called for this long, in ms:
; the engine is inside LoadMap and calling one there crashes the game.
reader_transition_cooldown_ms = 2000

; A position jump bigger than this inside one pump is a fast travel: the pawn is
; re-resolved and the widgets re-swept. Sprinting covers ~70 uu per pump.
reader_teleport_jump_uu = 3000

; How often the streamed level set is walked to name the chapter, in ms.
reader_chapter_period_ms = 1000

; Rate limit on the reader's repeating log lines, in ms.
reader_log_throttle_ms = 5000

; ---------------------------------------------------------------------------------
; MARKER SWEEP INTERNALS
; ---------------------------------------------------------------------------------
; Rounds a live actor may go unseen before it leaves the live cache. This is not a
; collected test - absence only ever marks through the markers_absence_* rules.
markers_live_grace_rounds = 2

; ---------------------------------------------------------------------------------
; ASSETS AND DIAGNOSTICS
; ---------------------------------------------------------------------------------
; How long a retired chapter's height planes stay alive after the pointer to them is
; cleared, in ms, so a slice already in flight cannot be handed freed memory.
map_asset_retire_grace_ms = 2000

; Rate limit on the "hidden because: ..." log line, in ms.
hide_reason_log_ms = 2000

; Our SRV descriptor heap: the ImGui font atlas, the map texture and the four slice
; buffers. RESTART ONLY.
srv_heap_size = 64

; Extra widget class-name prefixes that never count as an open menu, on top of the

; some transient combat furniture that way. Every root discovered for the first time
; is logged as "menu root discovered: ...", which is the name to put here.
; Comma-separated, matched case-insensitively; empty = built-in table only.
menu_ignore_roots =

; ---------------------------------------------------------------------------------
; THE X-RAY CAMERA READER
; ---------------------------------------------------------------------------------
; How often the camera manager is re-found when missing; the slower rate used when

; when the raw POV block could not be pinned. All in ms.
highlight_camera_resolve_ms = 500
highlight_compass_period_ms = 50
highlight_getter_period_ms = 33

; How far into CameraCachePrivate the POV block is looked for, in bytes, and how many
; insane reads in a row drop the pinned offset and re-run the discovery.
highlight_pov_scan_bytes = 192
highlight_pov_bad_reads = 8
